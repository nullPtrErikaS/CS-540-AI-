       .ORIG x3000
        AND R1, R1, #0      ; Clear R1 to use as a counter
        ADD R1, R1, #8      
        LD  R2, NEGZ
LOOP1   LDI R3, KBSR
        BRzp LOOP1          ; Loop until a key is pressed (KBSR bit [15] = 1)
        LDI R0, KBDR        ; Load the key pressed from KBDR into R0
        ADD R4, R0, R2      ; Add R2 (negation of 'Z') to character in R0
        BRnz NEXT            ; If the result is negative, character is less than 'Z', go to NEXT

LOOP2   LDI R5 DSR
        BRzp LOOP2          ; Loop until the display is ready
        STI R0, DDR         ; Store the character in R0 to DDR to display it

NEXT    ADD R1, R1, #-1 
        BRp LOOP1           ; If counter is not zero, repeat loop
        HALT                ; Stop the program

; Constants for device registers
NEGZ    .FILL xFFA6     ; NEGZ = -'Z' (-90 in decimal)
KBSR    .FILL xFE00     ; Keyboard status register address
KBDR    .FILL xFE02     ; Keyboard data register address
DSR     .FILL xFE04     ; Display status register address
DDR     .FILL xFE06     ; Display data register address
    .END

	LD R1, NEG0                 ; Load into R1 Line

	STI R0, DDRaddr
